CREATE OR REPLACE PACKAGE BODY NBH_CDS_NEW.MA_KPI_GLI_TONG AS

   PROCEDURE TONG_HOP_KPI(P_TUNGAY VARCHAR2, P_DENNGAY VARCHAR2,P_RESULT OUT SYS_REFCURSOR)
   IS
   P_THANG varchar2(20);
    BEGIN
    P_THANG:=to_char(to_date(P_TUNGAY,'dd/mm/yyyy'),'yyyymm');
    OPEN P_RESULT FOR
    SELECT TEN_KPI,
            SUM(KEHOACH_THANG)KEHOACH_THANG,
            SUM(KEHOACH_THANG_LK)KEHOACH_THANG_LK,
            SUM(THUCHIEN_THANG)THUCHIEN_THANG,
            SUM(THUCHIEN_THANG_LK)THUCHIEN_THANG_LK,
            ROUND(SUM(THUCHIEN_THANG)/ NULLIF(SUM(KEHOACH_THANG), 0) * 100,2) AS KETQUA_THANG,
            ROUND(SUM(THUCHIEN_THANG_LK)/ NULLIF(SUM(KEHOACH_THANG_LK), 0) * 100,2) AS KETQUA_LK_THANG
    FROM GLI_DM_KPI A
    JOIN GLI_KQ_KPI_TONG B ON A.MA_KPI=B.MA_KPI AND THANG=P_THANG
        GROUP BY TEN_KPI;
 END TONG_HOP_KPI;

END MA_KPI_GLI_TONG;
