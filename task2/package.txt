     ---- Lấy tổng hợp PTM theo nhân viên phát triển
PROCEDURE TONGHOP_PTM_NV (P_FROM IN VARCHAR2, P_TO IN VARCHAR2,P_DONVI  IN NUMBER,P_DICHVU IN NUMBER,P_LOAITB IN NUMBER,P_CURSOR OUT SYS_REFCURSOR)
    IS
        P_TUNGAY DATE:= TO_DATE(P_FROM,'yyyymmdd');
        P_DENNGAY DATE:= TO_DATE(P_TO, 'yyyymmdd');

    BEGIN

    OPEN P_CURSOR FOR
        SELECT CASE WHEN GROUPING(TEN_DV) = 1 THEN 'TỔNG' ELSE TEN_DV END AS "Tên đơn vị phát triển",
            SUM(CASE WHEN DICHVUVT_ID IN (1,10,11) THEN 1 ELSE 0 END)"Cố Định",
            SUM(CASE WHEN LOAITB_ID=58 THEN 1 ELSE 0 END)"Fiber",
            SUM(CASE WHEN LOAITB_ID=61 THEN 1 ELSE 0 END)"Mytv",
            SUM(CASE WHEN LOAITB_ID=210 THEN 1 ELSE 0 END)"Mesh",
            SUM(CASE WHEN DICHVUVT_ID IN (7,8,9) THEN 1 ELSE 0 END)"TSL",
            SUM(CASE WHEN DICHVUVT_ID IN (12) THEN 1 ELSE 0 END)"Camera/MegaEyes",
            SUM(CASE WHEN DICHVUVT_ID IN (13,14,15,16) THEN 1 ELSE 0 END)"Nhóm CNTT"
        FROM DB_PTM
            WHERE NGAY_HT >= P_TUNGAY AND NGAY_HT < P_DENNGAY+1 AND (P_DONVI = 0 OR DONVI_ID = P_DONVI)
            AND (P_DICHVU = 0 OR DICHVUVT_ID = P_DICHVU) AND (P_LOAITB = 0 OR LOAITB_ID = P_LOAITB)
          GROUP BY ROLLUP(TEN_DV) ORDER BY GROUPING(TEN_DV),TEN_DV;

    END TONGHOP_PTM_NV;

        -- Dòng tổng

        SUM(CASE WHEN LOAITB_ID = 58 THEN 1 ELSE 0 END) AS "Fiber",
        SUM(CASE WHEN LOAITB_ID = 61 THEN 1 ELSE 0 END) AS "Mytv",
        SUM(CASE WHEN LOAITB_ID = 210 THEN 1 ELSE 0 END) AS "Mesh",
        SUM(CASE WHEN DICHVUVT_ID IN (7,8,9) THEN 1 ELSE 0 END) AS "TSL",
        SUM(CASE WHEN DICHVUVT_ID = 12 THEN 1 ELSE 0 END) AS "Camera/MegaEyes",
        SUM(CASE WHEN DICHVUVT_ID IN (13,14,15,16) THEN 1 ELSE 0 END) AS "Nhóm CNTT",
        1 AS SORT_ORDER
    FROM DB_PTM
    WHERE NGAY_HT >= P_TUNGAY AND NGAY_HT < P_DENNGAY + 1 AND (P_DONVI = 0 OR DONVI_DB = P_DONVI)
        AND (P_KHUVUC = 0 OR KHUVUC_ID = P_KHUVUC) AND (P_DICHVU = 0 OR DICHVUVT_ID = P_DICHVU)
        AND (P_LOAITB = 0 OR LOAITB_ID = P_LOAITB)
    ORDER BY SORT_ORDER,"Tên đơn vị địa bàn","Tên khu vực","Mã nhân viên";

    END TONGHOP_PTM_KV;

         ---- Lấy tổng hợp thuê bao XGSPON theo địa bàn
    PROCEDURE TONGHOP_XGSPON_DB (P_FROM IN VARCHAR2, P_TO IN VARCHAR2,P_DONVI  IN NUMBER,P_DICHVU IN NUMBER,P_LOAITB IN NUMBER,P_CURSOR OUT SYS_REFCURSOR)
    IS
        P_TUNGAY DATE:= TO_DATE(P_FROM,'yyyymmdd');
        P_DENNGAY DATE:= TO_DATE(P_TO, 'yyyymmdd');

    BEGIN

    OPEN P_CURSOR FOR
        SELECT CASE WHEN GROUPING(TEN_DV_DB) = 1 THEN 'TỔNG' ELSE TEN_DV_DB END AS "Tên đơn vị phát triển",
            COUNT(*) AS "Tổng XGSPON",
            SUM(CASE WHEN XGSPON='PTM XGSPON' THEN 1 ELSE 0 END) "PTM XGSPON",
            SUM(CASE WHEN XGSPON='CHUYỂN ĐỔI XGSPON' THEN 1 ELSE 0END) "CHUYỂN ĐỔI XGSPON"
        FROM DB_XGSPON
            WHERE NGAY_HT >= P_TUNGAY AND NGAY_HT < P_DENNGAY+1 AND (P_DONVI = 0 OR DONVI_ID_DB = P_DONVI)
           GROUP BY ROLLUP(TEN_DV_DB) ORDER BY GROUPING(TEN_DV_DB),
          TEN_DV_DB;
    END TONGHOP_XGSPON_DB;
