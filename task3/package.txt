CREATE OR REPLACE PACKAGE BODY NBH_CDS_NEW.PACK_CNTT
IS
PROCEDURE TONGHOP_DOANHTHU_CNTT(P_FROM IN VARCHAR2, P_TO IN VARCHAR2,P_DONVI  IN NUMBER,P_DICHVU IN NUMBER,P_LOAITB IN NUMBER,P_CURSOR OUT SYS_REFCURSOR)
    IS
    P_THANG VARCHAR2(6);
BEGIN

     P_THANG := TO_CHAR(TO_DATE(P_FROM, 'DD/MM/YYYY'),'YYYYMM');
    OPEN P_CURSOR FOR
WITH DATA AS
(
    SELECT
        NVL(A.DONVI_ID, B.DONVI_ID) AS DONVI_ID,
        NVL(A.TEN_DV, B.TEN_DV) AS TEN_DV,
        B.KE_HOACH,A.DT_A,A.DT_B,A.DT_ABC,A.DT_TBI_A,A.DT_TBI_BC,A.DT_TBI_ABC,A.TTONG_DT,
        ROUND(A.TTONG_DT / NULLIF(B.KE_HOACH, 0) * 100,2) AS TY_LE
    FROM (
        SELECT DONVI_ID,MAX(TEN_DV) AS TEN_DV,
            ROUND(SUM(CASE WHEN KHOANMUCTT_ID = 11 AND NHOM_DV = 'Nhóm A' THEN TIEN ELSE 0 END)) AS DT_A,
            ROUND(SUM(CASE WHEN KHOANMUCTT_ID = 11 AND NHOM_DV = 'Nhóm B,C' THEN TIEN ELSE 0 END)) AS DT_B,
            ROUND(SUM(CASE WHEN KHOANMUCTT_ID = 11 THEN TIEN ELSE 0 END)) AS DT_ABC,
            ROUND(SUM(CASE WHEN KHOANMUCTT_ID = 5 AND NHOM_DV = 'Nhóm A' THEN TIEN ELSE 0 END)) AS DT_TBI_A,
            ROUND(SUM(CASE WHEN KHOANMUCTT_ID = 5 AND NHOM_DV = 'Nhóm B,C' THEN TIEN ELSE 0 END)) AS DT_TBI_BC,
            ROUND(SUM(CASE WHEN KHOANMUCTT_ID = 5 THEN TIEN ELSE 0 END )) AS DT_TBI_ABC,
            ROUND(SUM(TIEN)) AS TTONG_DT
        FROM
        (SELECT DECODE(DONVI_ID,314580, 1005145,1000638, 1005150,DONVI_ID) AS DONVI_ID,TEN_DV,KHOANMUCTT_ID,NHOM_DV,TIEN
            FROM GLI_THUEBAO_TRATRUOC
            WHERE DICHVUVT_ID IN (13,14,15,16) AND NGAY_HT >= TO_DATE(P_THANG || '01', 'YYYYMMDD')AND NGAY_HT < ADD_MONTHS( TO_DATE(P_THANG || '01', 'YYYYMMDD'),1))
        GROUP BY DONVI_ID) A

    FULL OUTER JOIN
    (SELECT DONVI_ID, MAX(TEN_DV) AS TEN_DV,
            SUM(KE_HOACH) AS KE_HOACH
        FROM GLI_KE_HOACH
        WHERE THANG = P_THANG
        GROUP BY DONVI_ID) B
        ON A.DONVI_ID = B.DONVI_ID)

SELECT
    DONVI_ID,TEN_DV,KE_HOACH,
    DT_A AS "DOANH THU NHÓM A",
    DT_B AS "DOANH THU NHÓM B,C",
    DT_ABC AS "DOANH THU",
    DT_TBI_A AS "DOANH THU TBỊ NHÓM A",
    DT_TBI_BC AS "DOANH THU TBỊ NHÓM B,C",
    DT_TBI_ABC AS "DOANH THU TBỊ",
    TTONG_DT AS "DT_THUCHIEN",
    TY_LE,
    0 STT
FROM DATA
UNION ALL
SELECT
    NULL AS DONVI_ID,
    'TỔNG' AS TEN_DV,
    SUM(KE_HOACH) AS KE_HOACH,
    SUM(DT_A) AS "DOANH THU NHÓM A",
    SUM(DT_B) AS "DOANH THU NHÓM B,C",
    SUM(DT_ABC) AS "DOANH THU",
    SUM(DT_TBI_A) AS "DOANH THU TBỊ NHÓM A",
    SUM(DT_TBI_BC) AS "DOANH THU TBỊ NHÓM B,C",
    SUM(DT_TBI_ABC) AS "DOANH THU TBỊ",
    SUM(TTONG_DT) AS "DT_THUCHIEN",
    ROUND(SUM(TTONG_DT) / NULLIF(SUM(KE_HOACH), 0)* 100,2) AS TY_LE,
    1 STT
FROM DATA
ORDER BY STT,TEN_DV;
 END TONGHOP_DOANHTHU_CNTT;

END PACK_CNTT;
